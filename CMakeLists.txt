cmake_minimum_required(VERSION 3.25)
project(TansrSDK VERSION 0.1.0 LANGUAGES CXX)

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(TANSR_BUILD_TESTS "Build SDK tests" ON)
option(TANSR_BUILD_DEMOS "Build public API demos" ON)
option(TANSR_BUILD_INTEGRATION "Build explicit real-Serve integration drivers" ON)
option(TANSR_WARNINGS_AS_ERRORS "Treat SDK warnings as errors" ON)
option(TANSR_ENABLE_ASAN "Enable address/undefined sanitizers" OFF)
option(TANSR_ENABLE_TSAN "Enable thread sanitizer (separate build)" OFF)
set(TANSR_CLANG_TIDY "" CACHE FILEPATH "Explicit clang-tidy executable")

if(TANSR_ENABLE_ASAN AND TANSR_ENABLE_TSAN)
  message(FATAL_ERROR "ASan and TSan require separate build trees")
endif()
find_package(CURL 8.0 REQUIRED)
find_package(OpenSSL 3.5 REQUIRED COMPONENTS Crypto)
if(TANSR_BUILD_TESTS)
  find_package(OpenSSL 3.5 REQUIRED COMPONENTS SSL)
endif()
find_package(Threads REQUIRED)

# 生成产物已入库；普通配置不运行生成器、不下载依赖。
file(GLOB_RECURSE TANSR_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")
add_library(tansr_sdk ${TANSR_SOURCES})
add_library(tansr::sdk ALIAS tansr_sdk)
set_target_properties(tansr_sdk PROPERTIES EXPORT_NAME sdk CXX_EXTENSIONS OFF
  VERSION ${PROJECT_VERSION} SOVERSION 0 POSITION_INDEPENDENT_CODE ON)
target_compile_features(tansr_sdk PUBLIC cxx_std_17)
# 公共头为UTF-8；这一读取要求必须随安装target传递给默认本地代码页的MSVC宿主。
target_compile_options(tansr_sdk INTERFACE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
target_include_directories(tansr_sdk PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(tansr_sdk PRIVATE CURL::libcurl OpenSSL::Crypto Threads::Threads)
target_compile_definitions(tansr_sdk PRIVATE TANSR_BUILDING_LIBRARY)
if(BUILD_SHARED_LIBS)
  target_compile_definitions(tansr_sdk PUBLIC TANSR_SHARED)
endif()
if(WIN32)
  target_compile_definitions(tansr_sdk PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(tansr_sdk PRIVATE advapi32 ws2_32)
endif()
function(tansr_build_rules target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
    if(TANSR_WARNINGS_AS_ERRORS)
      target_compile_options(${target} PRIVATE /WX)
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    if(TANSR_WARNINGS_AS_ERRORS)
      target_compile_options(${target} PRIVATE -Werror)
    endif()
  endif()
  if(TANSR_CLANG_TIDY)
    set_target_properties(${target} PROPERTIES CXX_CLANG_TIDY "${TANSR_CLANG_TIDY}")
  endif()
  if(TANSR_ENABLE_ASAN)
    if(MSVC)
      target_compile_options(${target} PRIVATE /fsanitize=address)
    else()
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  elseif(TANSR_ENABLE_TSAN)
    if(MSVC)
      message(FATAL_ERROR "TSan requires a supported Clang/GCC build")
    endif()
    target_compile_options(${target} PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=thread)
  endif()
endfunction()
tansr_build_rules(tansr_sdk)

if(TANSR_BUILD_TESTS)
  enable_testing()
  file(GLOB TANSR_TEST_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/tests/*_test.cpp")
  foreach(source IN LISTS TANSR_TEST_SOURCES)
    get_filename_component(name "${source}" NAME_WE)
    add_executable(${name} "${source}")
    target_link_libraries(${name} PRIVATE tansr::sdk Threads::Threads)
    if(name STREQUAL "runtime_test")
      target_link_libraries(${name} PRIVATE CURL::libcurl)
    elseif(name STREQUAL "crypto_test")
      target_link_libraries(${name} PRIVATE OpenSSL::Crypto)
    elseif(name STREQUAL "tls_test")
      target_link_libraries(${name} PRIVATE CURL::libcurl OpenSSL::SSL OpenSSL::Crypto)
    endif()
    target_compile_definitions(${name} PRIVATE TANSR_CONTRACT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/contract")
    if(WIN32)
      target_link_libraries(${name} PRIVATE ws2_32 advapi32)
      target_compile_definitions(${name} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    endif()
    tansr_build_rules(${name})
    add_test(NAME ${name} COMMAND ${name} "${CMAKE_CURRENT_SOURCE_DIR}/contract")
    set_tests_properties(${name} PROPERTIES TIMEOUT 120)
    if(name STREQUAL "terminal_persistence_test")
      add_test(NAME terminal_persistence_receipts COMMAND ${name} --receipts)
      add_test(NAME terminal_persistence_delta COMMAND ${name} --delta)
      add_test(NAME terminal_persistence_budget COMMAND ${name} --budget)
      set_tests_properties(terminal_persistence_receipts terminal_persistence_delta PROPERTIES TIMEOUT 600)
    endif()
  endforeach()
endif()

if(TANSR_BUILD_INTEGRATION)
  file(GLOB TANSR_INTEGRATION_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/*.cpp")
  foreach(source IN LISTS TANSR_INTEGRATION_SOURCES)
    get_filename_component(name "${source}" NAME_WE)
    add_executable(serve_${name} "${source}")
    target_link_libraries(serve_${name} PRIVATE tansr::sdk Threads::Threads)
    if(WIN32)
      target_compile_definitions(serve_${name} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    endif()
    tansr_build_rules(serve_${name})
  endforeach()
endif()

if(TANSR_BUILD_DEMOS AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/demo/CMakeLists.txt")
  add_subdirectory(demo)
endif()

  DESTINATION ${CMAKE_INSTALL_DATADIR}/TansrSDK/packaging COMPONENT SDK)
install(DIRECTORY packaging/licenses/ DESTINATION ${CMAKE_INSTALL_DATADIR}/TansrSDK/packaging/licenses COMPONENT SDK)

install(FILES "${CMAKE_CURRENT_BINARY_DIR}/TansrSDKConfig.cmake"
  "${CMAKE_CURRENT_BINARY_DIR}/TansrSDKConfigVersion.cmake"
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/TansrSDK COMPONENT SDK)
