cmake_minimum_required(VERSION 3.25)
project(TansrExternalConsumer LANGUAGES CXX)
find_package(TansrSDK 0.1 CONFIG REQUIRED)
add_executable(tansr-consumer main.cpp)
target_link_libraries(tansr-consumer PRIVATE tansr::sdk)
target_compile_features(tansr-consumer PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(tansr-consumer PRIVATE /W4 /WX /permissive- /utf-8)
else()
  target_compile_options(tansr-consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME installed-package-consumer COMMAND tansr-consumer)
