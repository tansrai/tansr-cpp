add_library(tansr_demo_common STATIC common.cpp)
target_link_libraries(tansr_demo_common PUBLIC tansr::sdk Threads::Threads)
target_include_directories(tansr_demo_common PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(WIN32)
  target_link_libraries(tansr_demo_common PRIVATE shell32)
endif()
tansr_build_rules(tansr_demo_common)
foreach(name IN ITEMS chat tools archive memory)
  add_executable(tansr-${name} ${name}.cpp)
  target_link_libraries(tansr-${name} PRIVATE tansr_demo_common)
  tansr_build_rules(tansr-${name})
  install(TARGETS tansr-${name} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} COMPONENT Demos)
endforeach()
